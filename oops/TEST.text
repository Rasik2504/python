class method
class Student:

    college = "ABC"

    @classmethod
    def change_college(cls, new_name):
        cls.college = new_name


Student.change_college("XYZ")

print(Student.college)

static method
class Math:

    @staticmethod
    def square(x):
        return x * x

print(Math.square(6))
